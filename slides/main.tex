\begin{frame}{Motivation}
    \begin{itemize}
        \item Reading strings of $\lambda$-expressions and evaluating them is itself a computational task.
        \item<2-> Church-Turing thesis: Any computatable task can be performed by a $\lambda$-expression.
        \item <3-> $\Rightarrow$ Lambda evaluation should also be possible with a $\lambda$-expression!
    \end{itemize}

    \vspace{2em}
    \centering
    \onslide<4-> $\therefore$ Goal: Parse and evaluate $\lambda$-expressions using $\lambda$-calculus!
    
    \begin{table}[]
        \begin{tabular}{rcl}
            \onslide<5->{String ``($\lambda$. 0)''} &
            \onslide<6->{$\xrightarrow[(\text{OCaml})]{\text{Tokenize}}$ &
                \texttt{[LPAR; LAM; NAT(0); RPAR]}} \\ &
            \onslide<7->{$\xrightarrow[(\text{OCaml})]{\text{Encode tokens}}$ &
                $\concrep{(\lambda.\; 0)}$} \\ &
            \onslide<8->{$\xrightarrow{\text{Parser}}$ &
                $\text{App} (
                    \texttt{e\_parse},
                    \concrep{(\lambda.\; 0)}
                ) =_\beta \absrep{(\lambda.\; 0)}$} \\ &
            \onslide<9-> {$\xrightarrow{\text{Evaluator}}$ &
                $\text{App} (
                    \texttt{e\_eval},
                    \absrep{(\lambda.\; 0)}
                ) =_\beta (\lambda.\; 0)$}
        \end{tabular}
    \end{table}
\end{frame}

            \textbf{Constructor} & & \textbf{Scott Encoding} \\

\begin{frame}{Theory: Church vs. Scott Encoding}
    \begin{itemize}
        \item General Church encoding for ADTs: Consume with fold operations
        \item $\leftrightarrow$ Scott encoding: match expressions
    \end{itemize}

    \begin{table}[]
        \centering
        \small
        \begin{tabular}{@{}lll@{}}
            \onslide<2->{%
            \textbf{Constructor} & \textbf{Church} & \textbf{Scott} \\
            \texttt{nil} & $\lambda n\, c.\; n$ & $\lambda n\, c.\; n$ \\
            $\texttt{cons}\; h\; t$ & $\lambda n\, c.\; c\; h\; (t\; n\; c)$ & $\lambda n\, c.\; c\; h\; t$ \\
            \hline}\onslide<3->{%
            \noindent%
            $\text{Var}(n)$ & $\lambda v\, l\, a.\; v\; n$ & $\lambda v\, l\, a.\; v\; n$ \\
            $\text{Lam}(t)$ & $\lambda v\, l\, a.\; l\; (t\; v\; l\; a)$ & $\lambda v\, l\, a.\; l\; t$ \\
            $\text{App}(t_1, t_2)$ & $\lambda v\, l\, a.\; a\; (t_1\; v\; l\; a)\; (t_2\; v\, l\; a)$ & $\lambda v\, l\, a.\; a\; t_1\; t_2$ \\
            }
        \end{tabular}
    \end{table}

    \begin{itemize}
        \item<4-> Church: Top-level $v, l, a$ are passed to every subexpression (Bottom-up)
        \item<4-> Scott: Top-level $v, l, a$ operate on the immediate subexpressions directly (Top-down)
        \item<5-> For parsing and evaluating, \textbf{Scott encoding} is more suitable!
    \end{itemize}
\end{frame}

\begin{frame}{Self-Interpreter \texttt{e\_eval}}
    \begin{itemize}
        \item Interpreter \texttt{e\_eval} must satisfy $\text{App} (\texttt{e\_eval}, \absrep{M}) =_\beta M$.
        \item<2-> Conceptually, $\texttt{eval} \in
            \texttt{AST} \times \texttt{Env} \rightarrow \lambda\texttt{-expr.}$
            \item<3-> $\texttt{eval}\; \absrep{n}\; env = (n\text{-th variable in }env)$
            \item<4-> $\texttt{eval}\; \absrep{\lambda x.\; t}\; env =
                \lambda x.\; (\texttt{eval}\; t\; (x :: env))$
            \item<5-> $\texttt{eval}\; \absrep{t_1\; t_2}\; env =
                (\texttt{eval}\; t_1\; env)\; (\texttt{eval}\; t_2\; env)$
    \end{itemize}
    \begin{align*}
        \onslide<6->{
        \texttt{eval\_comb}
        &= \lambda\; ast\, lst.\; ast \\
        &\qquad (\lambda n.\; \texttt{lookup}\; lst\; n)
            \qquad\qquad\qquad\qquad\qquad\qquad\quad\;\; (\text{Var: Lookup})\\
        &\qquad (\lambda t.\; \texttt{eval\_comb}\; t\; (\texttt{cons}\; lst\; (\text{Var}\; 0)))
            \qquad\qquad\qquad (\text{Lam: Define variable, evaluate}) \\
        &\qquad (\lambda t_1\, t_2.\; (\texttt{eval\_comb}\; t_1\; lst)\; (\texttt{eval\_comb}\; t_2\; lst))
            \qquad (\text{App: Evaluate each, apply})
        }
    \end{align*}
    \begin{itemize}
        \item<8-> \texttt{cons}: Push every index up by one
        \item<8-> Recursion via $Y$-combinator
        \item<9-> $\Rightarrow\texttt{e\_eval}
            = \lambda\; ast.\; \texttt{eval\_comb} \; ast \; \texttt{nil}$
    \end{itemize}

\end{frame}

\begin{frame}{Self-Parser \texttt{e\_parse}}
    \begin{itemize}
        \item Similar for parsing, though more complicated case analysis
        \item<2-> $\texttt{parse} \in \texttt{tok list} \rightarrow \texttt{AST} \times \texttt{tok list}$
    \end{itemize}
    \begin{align*}
        \onslide<3-> {\texttt{parse}\; (\concrep{\texttt{NAT(n)}} :: lst)
            =&\; (\absrep{\text{Var}(n)}, lst)} \\
        \onslide<4-> {\texttt{parse}\; (\concrep{\texttt{LPAR}} :: \concrep{\texttt{LAM}} :: lst)
            =&\; \texttt{let}\; (t,\; \concrep{\texttt{RPAR}} :: lst') = \texttt{parse}\; lst\; \texttt{in} \\
             &\; (\absrep{\text{Lam}(t)}, lst')} \\
        \onslide<5-> {\texttt{parse}\; (\concrep{\texttt{LPAR}} :: lst)
            =&\; \texttt{let}\; (t_1,\; \concrep{\texttt{RPAR}} :: lst')
                = \texttt{parse}\; lst\; \texttt{in} \\
             &\; \texttt{let}\; (t_2,\; \concrep{\texttt{RPAR}} :: lst'')
                = \texttt{parse}\; lst'\; \texttt{in} \\
             &\; (\absrep{\text{App}(t_1, t_2)}, lst'')}
    \end{align*}

        \item<6-> Implementation in GitHub repository!
    \end{itemize}
\end{frame}
