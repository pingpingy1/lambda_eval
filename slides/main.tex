\begin{frame}{Motivation}
    \begin{itemize}
        \item Reading strings of $\lambda$-expressions and evaluating them is itself a computational task.
        \item<2-> Church-Turing thesis: Any computatable task can be performed by a $\lambda$-expression.
        \item <3-> $\Rightarrow$ Lambda evaluation should also be possible with a $\lambda$-expression!
    \end{itemize}

    \vspace{2em}
    \centering
    \onslide<4-> $\therefore$ Goal: Parse and evaluate $\lambda$-expressions using $\lambda$-calculus!
\end{frame}

        \onslide<3->{
        \begin{align*}
            t
            ::=&\; \text{Var}(n) & (\text{Variable}) \\
            \mid&\; \text{Lam}(t) & (\text{Function}) \\
            \mid&\; \text{App}(t, t) & (\text{Application})
        \end{align*}
        }
    \end{columns}

    \vspace{2em}
    \centering
    \onslide<4->{
    e.g.) ``($\lambda$. 0)''
    $\rightarrow$ \texttt{[LPAR; LAM; NAT(0); LPAR]}
    $\rightarrow$ $\lambda$-repr. of tokens
    $\rightarrow$ $\lambda$-repr. of AST
    $\rightarrow$ $\lambda$-expr.
    }

    \onslide<5-> {
    \begin{enumerate}
        \item Represent each token
        \item Represent each AST constructor
        \item Parser \texttt{e\_parse} from token list to AST
        \item Evaluator \texttt{e\_eval} from AST to $\lambda$-expression
    \end{enumerate}
    }
\end{frame}

\begin{frame}{Theory: Algebraic Data Types and Scott Encoding}
    \begin{itemize}
        \item Algebraic data type (ADT, 조립식 타입): Sum, product, and recursive types
        \item<2-> Scott encoding: General encoding of ADTs
    \end{itemize}

    \begin{columns}
    \column{0.48\textwidth}
    \onslide<3-> {
    \begin{table}[]
        \begin{tabular}{ll}
            \textbf{Constructor} & \textbf{Scott Encoding} \\
            $\text{Var}(n)$ & $\lambda v\, l\, a.\; v\; n$ \\
            $\text{Lam}(t)$ & $\lambda v\, l\, a.\; l\; t$ \\
            $\text{App}(t_1, t_2)$ & $\lambda v\, l\, a.\; a\; t_1\; t_2$ \\
            \hline
            \texttt{LPAR}   & $\lambda l\, r\, a\, v.\; l$ \\
            \texttt{RPAR}   & $\lambda l\, r\, a\, v.\; r$ \\
            \texttt{LAM}    & $\lambda l\, r\, a\, v.\; a$ \\
            \texttt{NAT(n)} & $\lambda l\, r\, a\, v.\; v\; n$ \\
        \end{tabular}
    \end{table}
    }

    \column{0.48\textwidth}
    \onslide<4-> {
    \textbf{Case statement}
    \[
        t\; fv\; fl\; fa = \begin{cases}
            fv\; n         & (t = \text{Var}(n)) \\
            fl\; t'        & (t = \text{Lam}(t')) \\
            fa\; t_1\; t_2 & (t = \text{App}(t_1, t_2)
        \end{cases}
    \]
    }
    \end{columns}

    \onslide<5-> {
    \begin{exampleblock}{Example: $\lambda x. \lambda y. x$}
        \[
            \lambda v\, l\, a.\; l\; \onslide<6-> {
                (\lambda v\, l\, a.\; l\; \onslide<7-> {
                    (\lambda v\, l\, a.\; a\; \onslide <8-> {
                        (\lambda s\, z.\; s\; z)
                    })
                })
            }
        \]
    \end{exampleblock}
    }
\end{frame}

\begin{frame}{Theory: Church vs. Scott Encoding}
    \begin{itemize}
        \item General Church encoding for ADTs: Consume with fold operations
        \item $\leftrightarrow$ Scott encoding: match expressions
    \end{itemize}

    \begin{table}[]
        \centering
        \small
        \begin{tabular}{@{}lll@{}}
            \onslide<2->{
            \textbf{Constructor} & \textbf{Church} & \textbf{Scott} \\
            $0$ & $\lambda s\, z.\; z$ & $\lambda s\, z.\; z$ \\
            $\texttt{succ } n$ & $\lambda s\, z.\; s\; z$ & $\lambda s\, z.\; s\; (\lambda s\, z.\; z)$ \\
            \hline}\onslide<3->{%
            \noindent%
            $\text{Var}(n)$ & $\lambda v\, l\, a.\; v\; n$ & $\lambda v\, l\, a.\; v\; n$ \\
            $\text{Lam}(t)$ & $\lambda v\, l\, a.\; l\; (t\; v\; l\; a)$ & $\lambda v\, l\, a.\; l\; t$ \\
            $\text{App}(t_1, t_2)$ & $\lambda v\, l\, a.\; a\; (t_1\; v\; l\; a)\; (t_2\; v\, l\; a)$ & $\lambda v\, l\, a.\; a\; t_1\; t_2$ \\
            }
        \end{tabular}
    \end{table}

    \begin{itemize}
        \item<4-> Church: Top-level $v, l, a$ are passed to every subexpression (Bottom-up)
        \item<4-> Scott: Top-level $v, l, a$ operate on the immediate subexpressions directly (Top-down)
        \item<5-> For parsing and evaluating, \textbf{Scott encoding} is more suitable!
    \end{itemize}
\end{frame}

\begin{frame}{Self-Interpreter \texttt{e\_eval}}
    \begin{itemize}
        \item Suppose $\ulcorner M \urcorner$ is the AST encoding of a $\lambda$-expression $M$.
        \item<2-> Interpreter \texttt{e\_eval} must satisfy $\text{App} (\texttt{e\_eval}, \ulcorner M \urcorner) \simeq_\beta M$.
    \end{itemize}
    \begin{align*}
        \onslide<3->{
        \texttt{e\_eval}
        &= \lambda\; ast.\; \texttt{eval\_comb} \; ast \; \texttt{nil}
        \qquad (\text{List of variables that have been defined})\\
        }
        \onslide<4->{
        \texttt{eval\_comb}
        &= \lambda\; ast\, lst.\; ast \\
        &\qquad (\lambda n.\; \texttt{lookup}\; lst\; n)
            \qquad\qquad\qquad\qquad\qquad\qquad\quad\;\; (\text{Var: Lookup})\\
        &\qquad (\lambda t.\; \texttt{eval\_comb}\; t\; (\texttt{cons}\; lst\; (\text{Var}\; 0)))
            \qquad\qquad\qquad (\text{Lam: Define variable, evaluate}) \\
        &\qquad (\lambda t_1\, t_2.\; (\texttt{eval\_comb}\; t_1\; lst)\; (\texttt{eval\_comb}\; t_2\; lst))
            \qquad (\text{App: Evaluate each, apply})
        }
    \end{align*}
    \begin{itemize}
        \item<5-> \texttt{cons}: Push every index up by one
        \item<5-> Recursion via $Y$-combinator
    \end{itemize}

\end{frame}
